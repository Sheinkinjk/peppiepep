# Note on Disallow vs noindex: a page must be CRAWLABLE for Google to read its
# noindex tag. Disallowing a page that also carries noindex means Google can never
# see the directive, so an already-indexed URL can never be dropped. Pages below
# that carry noindex in their metadata are therefore deliberately NOT disallowed
# here: /login, /roi-calculator (301), /integrations + children.
#
# Pages that return 410 Gone (the retired SaaS: /status, /go-live, /analytics,
# /calendly, /dashboard-test, the integration docs) are also left crawlable, so
# crawlers can read the 410 and drop them.
#

Allow: /
# Private / authenticated (no content to index; auth-redirects before render)
Disallow: /dashboard/
Disallow: /api/
Disallow: /auth/
# Internal / utility pages (dev checklists, tools - no public SEO value)
Disallow: /application
# Personalised referral links (redirect out to affiliate destinations)
Disallow: /go/
Disallow: /r/
Disallow: /me/
# Payment flows
Disallow: /payment/
